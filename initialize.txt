## Create New Project
    git init
    git add  README.md
    git commit -m "First commit"
    git branch -M main
    git remote add origin https://path.git
    git push -u origin main




## Start Project
    git init
    git status
    git add .
    git commit -m "First commit"

    git config --global user.name "Your Name"
    git config --global user.email you@example.com
    git commit --amend --reset-author --no-edit

    git diff  (q --> for end)
    git log --oneline



## Create a New Brach
    git switch -c feature/new-command
    git branch
    git status
    git add .
    git commit -m " "
    git status
    git log --oneline

## Merge branch
    git switch master
    git merge feature/delete-task
    git log --oneline --graph --all
