## Create New Project
    git init
    git add  README.md
    git commit -m "First commit"
    git branch -M main  (-M --> --move --force)
    git remote add origin https://path.git
    git push -u origin main  (-u --> --upstream)




## Start Project
    git clone https://github.com/user/project.git (This command is for first time)


    git init
    git remote add origin https://path.git
    git branch -r (--remote or -a --> --all)
    git checkout -b master origin/master
    git status
    git add .
    git commit -m "First commit"
    git fetch origin (For any times that we want to get the last changes on the project)

    git config --global user.name "Your Name"
    git config --global user.email you@example.com
    git commit --amend --reset-author --no-edit

    git diff  (q --> for end)
    git log --oneline



## Create a New Brach
    git switch -c feature/new-command
    git branch --show-current OR git branch
    git status
    git add .
    git commit -m " "
    git status
    git log --oneline

## Merge branch
    git switch master
    git merge feature/delete-task
    git log --oneline --graph --all 


## Push on server
    git remote add origin <REPOSITORY_URL>
    git remote -v
    git push -u origin [branch_name] (-u --> --upstream)
    git pull

## Remove unstage changes
    git restore <file>

## Commit New Changes that remove the mentioned commit
    git log --oneline
    git revert <commit> (--> to close the file :wq)


## Save unstage changes and use it again
    git stash
    git switch main
    git stash pop

## History
    git show head
    git show head~1...N
    git revert head
    git revert head~1...N
